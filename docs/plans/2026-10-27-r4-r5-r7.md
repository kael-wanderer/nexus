    nonisolated static func includes(_ url: URL, excludingRoot: URL?) -> Bool {

- [ ] Add the following test inside `ProviderTests` in `Tests/NexusCoreTests/SearchEngineTests.swift`. This fake models prefix retrieval and applies the same exclusion before its limit. It deliberately includes more excluded hits than the provider's candidate budget. The ranking assertions exercise the actual provider and `Ranking.sorted`.

```swift
    @Test("File search examples exclude Library before the candidate limit")
    func fileSearchExamples() async throws {
        let home = FileManager.default.homeDirectoryForCurrentUser
        let library = home.appendingPathComponent("Library", isDirectory: true)
        let examples: [(String, String, String)] = [
            ("budget", "Budget.xlsx", "Documents/Finance"),
            ("invoice", "Invoice.pdf", "Downloads"),
            ("meeting", "Meeting.txt", "Documents"),
            ("project", "Project.swift", "Projects/App"),
            ("safari", "Safari notes.txt", "Desktop"),
            ("recipe", "Recipe.md", "Documents/Food"),
            ("photo", "Photo.jpg", "Pictures"),
            ("report", "Report.pdf", "Documents/Work"),
            ("design", "Design.fig", "Projects/Studio"),
            ("tax", "Tax.pdf", "Library Notes")
        ]
        for (query, expected, parent) in examples {
            let excluded = (0..<100).map { index in
                MetadataItem(url: library.appendingPathComponent("Caches/\(query)-\(index)"),
                             displayName: "\(query)-\(index)")
            }
            let wanted = MetadataItem(url: home.appendingPathComponent("\(parent)/\(expected)"),
                                      displayName: expected)
            let candidates = excluded + [wanted]
            let provider = FileSearchProvider(run: { text, limit, _ in
                Array(candidates.lazy.filter {
                    $0.displayName.lowercased().hasPrefix(text.lowercased())
                        && MetadataQueryRunner.includes($0.url, excludingRoot: library)
                }.prefix(limit))
            })
            let results = Ranking.sorted(await provider.results(
                for: SearchQuery(text: query, token: 1), context: SearchContext()
            ))
            try #require(results.count > 0)
            #expect(results[0].title == expected)
            #expect(results[0].action == .openFile(wanted.url))
            #expect(results[0].subtitle == "~/\(parent)")
            #expect(results.count == 1)
        }
        #expect(!MetadataQueryRunner.includes(library, excludingRoot: library))
        #expect(!MetadataQueryRunner.includes(library.appendingPathComponent("Caches/a"), excludingRoot: library))
        #expect(MetadataQueryRunner.includes(home.appendingPathComponent("Library Notes/a"), excludingRoot: library))
        #expect(MetadataQueryRunner.includes(library, excludingRoot: nil))
    }
```
